===============================================================
               Production environment setup
===============================================================

//To bring up all containers
docker compose up -d

//To bring down all the containers
docker compose down

//Commands to run manual service containers

To send test data :
docker compose run seed-data

To check redis and database reachability : 
docker compose run service_check

enter password when prompted, pwd: postgres
-----------------------------------------------------------------
    OR 
-----------------------------------------------------------------
Execute via docker scripts

cd docker-scripts/

sudo ./docker_prod.sh {up|down|build|test|health}

===============================================================
    Dev Environment Setup (live source coding + hot reload!)
===============================================================

docker compose -f docker-compose.yaml -f docker-compose.dev.yaml up
docker compose -f docker-compose.yaml -f docker-compose.dev.yaml down -v
docker compose -f docker-compose.yaml -f docker-compose.dev.yaml --build
----------------------------------------------------------------
    OR

Execute via docker scripts

cd docker-scripts/
